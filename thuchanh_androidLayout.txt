// RelativeLayout: sắp xếp các view tương đối với nhau hoặc với parent container.
<RelativeLayout
	>
	
	<TextView
        	android:id="@+id/label"
        />
	
	<EditText
		android:layout_above="@id/<id>"		// Đặt View bên trên 1 View khác
		android:layout_below="@id/<id>"		// Đặt View bên dưới 1 View khác
		android:layout_toStartOf="@id/<id>"	// Đặt View bên trái 1 View khác
		android:layout_toEndOf="@id/<id>"	// Đặt View bên phải 1 View khác
	
		android:layout_alignTop="@id/<id>"	// Căn mép trên của View theo View khác (giúp căn chỉnh View thẳng hàng, nếu ko sẽ bị lệch, đi chung với bottom để có height giống nhau)x`
		android:layout_alignBottom="@id/<id>"	// Căn mép dưới của View theo View khác 
		android:layout_alignLeft="@id/<id>"	// Căn mép trái của View theo View khác (thường đi chung với right để có width giống nhau)
		android:layout_alignRight="@id/<id>"	// Căn mép phải của View theo View khác

		android:layout_alignBaseline="@id/<id>"	// Căn baseline chữ của View với View khác (chung 1 dòng)
	/>
	
	<EditText
		android:layout_alignParentStart="true"	// căn View sát mếp trái
		android:layout_alignParentEnd="true"	// căn View sát mép phải
		android:layout_alignParentTop="true"	// Căn View sát mép trên
		android:layout_alignParentBottom="true"	// Căn View sát mép dưới
		android:layout_centerHorizontal="true"	// Căn View theo chiều ngang
		android:layout_centerVertical="true"	// Căn giữa View theo chiều dọc
		android:layout_centerInParent="true"	// Căn giữa View cả chiều ngang và chiều dọc
	/>
</RelativeLayout>	


// ConstraintLayout: 1 layout mạnh, giúp thiết kế giao diện linh hoạt và tối ưu hiệu suất. Cho phép ràng buộc, không cần lồng nhiều layout RelativeLayout hay LinearLayout

androidx.constraintlayout.widget.ConstraintLayout

<ConstraintLayout
	>

	<Button
		tools:layout_editor_absoluteX="15dp"	// Xuất hiện tại vị trí có x = 15dp
		tools:layout_editor_absoluteY="30dp"	// Xuất hiện tại vị trí có x = 30dp
	/>

	<TextView
		android:hint="name@gmail.com"					// placeholder


		app:layout_constraintLeft_toLeftOf="@id/<id>"		// Ràng buộc cạnh trái của View với cạnh trái của View khác

		app:layout_constraintLeft_toRightOf="@id/<id>"		// Ràng buộc cạnh trái của View với cạnh phải của View khác

		app:layout_constraintRight_toLeftOf="@id/<id>"		// Ràng buộc cạnh phải của View với cạnh trái của View khác

		app:layout_constraintRight_toRightOf="@id/<id>"		// Ràng buộc cạnh phải của View với cạnh phải của View khác

		app:layout_constraintTop_toTopOf="@id/<id>"			// Ràng buộc cạnh trên của View với cạnh trên của View khác

		app:layout_constraintTop_toBottomOf="@id/<id>"		// Ràng buộc cạnh trên của View với cạnh dưới của View khác

		app:layout_constraintBottom_toTopOf="@id/<id>"		// Ràng buộc cạnh dưới của View với cạnh trên của View khác

		app:layout_constraintBottom_toBottomOf="@id/<id>"		// Ràng buộc cạnh dưới của View với cạnh dưới của View khác

		app:layout_constraintBaseline_toBaselineOf="@id/<id>"	// Ràng buộc baseline (đường gốc chữ) của View với baseline của View khác

		app:layout_constraintStart_toEndOf="@id/<id>"		// Ràng buộc bắt đầu (Start) của View với kết thúc (End) của View khác (hỗ trợ RTL)

		app:layout_constraintStart_toStartOf="@id/<id>"		// Ràng buộc bắt đầu (Start) của View với bắt đầu (Start) của View khác

		app:layout_constraintEnd_toStartOf="@id/<id>"		// Ràng buộc kết thúc (End) của View với bắt đầu (Start) của View khác

		app:layout_constraintEnd_toEndOf="@id/<id>"			// Ràng buộc kết thúc (End) của View với kết thúc (End) của View khác
	/>

</ConstraintLayout>
